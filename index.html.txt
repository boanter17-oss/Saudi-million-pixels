        <button class="cta-btn" onclick="bookPixel()">احجز مربعك برابط موقعك الآن</button>

    <script>
        const totalPixels = 200; // عدد المربعات الافتراضي (يمكنك زيادته لاحقاً)

        // قاعدة بيانات المربعات المحجوزة (هنا يظهر اسم المتجر ورابطه مباشرة داخل المربع)
        const reservedData = {
            5: { name: "متجر تقني", url: "https://x.com" }, 
            12: { name: "منصة ذكية", url: "https://google.com" }, 
            22: { name: "قهوة مختصة", url: "https://github.com" } 
        };

        const grid = document.getElementById('grid');
        const reservedKeys = Object.keys(reservedData).map(Number);
        
        // تحديث العدادات
        document.getElementById('reserved-count').innerText = reservedKeys.length;
        document.getElementById('remaining-count').innerText = totalPixels - reservedKeys.length;

        // بناء الشبكة
        for (let i = 1; i <= totalPixels; i++) {
            const pixel = document.createElement('div');
            pixel.classList.add('pixel');

            if (reservedKeys.includes(i)) {
                pixel.classList.add('reserved');
                const data = reservedData[i];
                pixel.title = `مربع #${i} - ${data.name} (اضغط لزيارة الموقع)`;
                pixel.innerHTML = `<span>${data.name}</span><small style="font-size:0.6rem; opacity:0.8;">#${i}</small>`;

                pixel.addEventListener('click', () => {
                    window.open(data.url, '_blank');
                });
            } else {
                pixel.classList.add('empty');
                pixel.title = `مربع رقم #${i} (متاح للحجز)`;
                pixel.innerHTML = `<span>#${i}</span>`;
                
                pixel.addEventListener('click', () => {
                    const confirmBook = confirm(`المربع رقم #${i} متاح للحجز!\nهل تريد التواصل لحجزه وإدراج اسم متجرك ورابطك فيه؟`);
                    if (confirmBook) {
                        bookPixelWithNumber(i);
                    }
                });
            }
            
            grid.appendChild(pixel);
        }

        function bookPixel() {
            const phone = "966500000000"; // استبدل برقم واتسابك
            const message = encodeURIComponent("أهلاً، أرغب في حجز مربع إعلاني على شبكة المليون برابط موقعي واسم متجري.");
            window.open(`https://wa.me/${phone}?text=${message}`, '_blank');
        }

        function bookPixelWithNumber(pixelNum) {
            const phone = "966500000000"; // استبدل برقم واتسابك
            const message = encodeURIComponent(`أهلاً، أرغب في حجز المربع رقم #${pixelNum} على شبكة المليون! اسم متجري هو: [اكتب اسمك] ورابط موقعي هو: [ضع رابطك]`);
            window.open(`https://wa.me/${phone}?text=${message}`, '_blank');
        }
    </script>
